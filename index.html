import SwiftUI import MultipeerConnectivity import Combine // MARK: - Main Container struct ContentView: View { @StateObject private var multiplayer = MultiplayerManager() @State private var isGameActive = true var body: some View { if isGameActive { GameView(multiplayer: multiplayer, isGameActive: $isGameActive) } else { VStack(spacing: 20) { Text("Main Menu") .font(.largeTitle) .bold() Button("Start Game") { isGameActive = true } .buttonStyle(.borderedProminent) } } } } // MARK: - Gameplay View struct GameView: View { @ObservedObject var multiplayer: MultiplayerManager @Binding var isGameActive: Bool @State private var isPaused = false @State private var isGameOver = false @State private var health = 100 var body: some View { ZStack { // Gameplay Layer VStack(spacing: 20) { // Header / Connection Status HStack { VStack(alignment: .leading) { Text("Health: \(health)") .font(.title2) .bold() .foregroundColor(health > 30 ? .green : .red) Text(multiplayer.statusText) .font(.caption) .foregroundColor(.secondary) } Spacer() Button("Pause") { isPaused = true } .buttonStyle(.bordered) } .padding() Spacer() // Testing Controls (Dying Mechanics) VStack(spacing: 12) { Text("Testing Controls") .font(.headline) HStack(spacing: 16) { Button("Take 25 Damage") { takeDamage(25) } .buttonStyle(.bordered) .tint(.orange) Button("Instant Die") { takeDamage(health) } .buttonStyle(.borderedProminent) .tint(.red) } } .padding() .background(Color(.secondarySystemBackground)) .cornerRadius(12) Spacer() } .blur(radius: (isPaused || isGameOver) ? 6 : 0) .disabled(isPaused || isGameOver) // Pause Overlay if isPaused { OverlayCard(title: "Game Paused") { Button("Resume") { isPaused = false } .buttonStyle(.borderedProminent) Button("Leave Game") { isPaused = false isGameActive = false } .buttonStyle(.bordered) .tint(.red) } } // Game Over Overlay if isGameOver { OverlayCard(title: "You Died!") { Button("Respawn") { health = 100 isGameOver = false } .buttonStyle(.borderedProminent) Button("Leave to Menu") { isGameOver = false isGameActive = false } .buttonStyle(.bordered) .tint(.red) } } } .onAppear { multiplayer.start() } .onDisappear { multiplayer.stop() } } private func takeDamage(_ amount: Int) { health = max(0, health - amount) if health == 0 { isGameOver = true } } } // MARK: - Reusable UI Overlay Component struct OverlayCard: View { let title: String @ViewBuilder let content: () -> Content var body: some View { ZStack { Color.black.opacity(0.4).ignoresSafeArea() VStack(spacing: 16) { Text(title) .font(.title) .bold() content() } .padding(24) .frame(maxWidth: 300) .background(RoundedRectangle(cornerRadius: 16).fill(Color(.systemBackground))) } } } // MARK: - Reliable Multipeer Manager class MultiplayerManager: NSObject, ObservableObject, MCSessionDelegate, MCNearbyServiceAdvertiserDelegate, MCNearbyServiceBrowserDelegate { private let serviceType = "pad-game" // Max 15 chars, lowercase, numbers, hyphens only private let myPeerId = MCPeerID(displayName: UIDevice.current.name) private var session: MCSession! private var advertiser: MCNearbyServiceAdvertiser! private var browser: MCNearbyServiceBrowser! @Published var statusText = "Searching for nearby iPads..." override init() { super.init() session = MCSession(peer: myPeerId, securityIdentity: nil, encryptionPreference: .required) session.delegate = self advertiser = MCNearbyServiceAdvertiser(peer: myPeerId, discoveryInfo: nil, serviceType: serviceType) advertiser.delegate = self browser = MCNearbyServiceBrowser(peer: myPeerId, serviceType: serviceType) browser.delegate = self } func start() { advertiser.startAdvertisingPeer() browser.startBrowsingForPeers() } func stop() { advertiser.stopAdvertisingPeer() browser.stopBrowsingForPeers() session.disconnect() } // Automatically accept incoming connections func advertiser(_ advertiser: MCNearbyServiceAdvertiser, didReceiveInvitationFromPeer peerID: MCPeerID, withContext context: Data?, invitationHandler: @escaping (Bool, MCSession?) -> Void) { invitationHandler(true, session) } // Automatically invite discovered peers func browser(_ browser: MCNearbyServiceBrowser, foundPeer peerID: MCPeerID, withDiscoveryInfo info: [String : String]?) { browser.invitePeer(peerID, to: session, withContext: nil, timeout: 10) } func browser(_ browser: MCNearbyServiceBrowser, lostPeer peerID: MCPeerID) {} func session(_ session: MCSession, peer peerID: MCPeerID, didChange state: MCSessionState) { DispatchQueue.main.async { switch state { case .connected: self.statusText = "Connected to \(peerID.displayName)" case .connecting: self.statusText = "Connecting..." case .notConnected: self.statusText = "Disconnected. Searching..." @unknown default: break } } } func session(_ session: MCSession, didReceive data: Data, fromPeer peerID: MCPeerID) {} func session(_ session: MCSession, didReceive stream: InputStream, withName streamName: String, fromPeer peerID: MCPeerID) {} func session(_ session: MCSession, didStartReceivingResourceWithName resourceName: String, fromPeer peerID: MCPeerID, with progress: Progress) {} func session(_ session: MCSession, didFinishReceivingResourceWithName resourceName: String, fromPeer peerID: MCPeerID, at localURL: URL?, withError error: Error?) {} }
